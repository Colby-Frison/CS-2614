/ code for odd sum
CLA                /clear

SUMLOOP, LDA SUM   /start loop load sum
    ADD NUM        /add current odd num to sum
    STA SUM        /store new sum

    LDA NUM        /load num
    ADD DECREMENT      /decrease by 2
    STA NUM        /sta decremented num
    LDA NUM        /reload num to check negative

    SNA            /if num is < 0 skip call back to loop
    BUN SUMLOOP

/convert to octal for output

/ Convert SUM to Octal
    LDA SUM         / Load the final sum
    STA DECIMAL     / Store it in DECIMAL for conversion

/ Calculate first digit (LSB)
    LDA DECIMAL     / Load current decimal value
    BSA DIVIDE_BY_8 / Call division subroutine
    LDA TEMP_REMAINDER / Load remainder
    STA OCTAL_RESULT0 / Store as least significant octal digit
    LDA QUOTIENT_TEMP  / Load quotient
    STA DECIMAL     / Update decimal value

/ Calculate second digit
    LDA DECIMAL
    BSA DIVIDE_BY_8
    LDA TEMP_REMAINDER
    STA OCTAL_RESULT1
    LDA QUOTIENT_TEMP
    STA DECIMAL

/ Calculate third digit
    LDA DECIMAL
    BSA DIVIDE_BY_8
    LDA TEMP_REMAINDER
    STA OCTAL_RESULT2
    LDA QUOTIENT_TEMP
    STA DECIMAL

/ Calculate fourth digit (MSB)
    LDA DECIMAL     / The remaining value is the most significant digit
    STA OCTAL_RESULT3 / Store it

OUTPUT_DONE, LDA OCTAL_RESULT3 / Start with most significant digit
    ADD CONST_48           / Convert to ASCII
    OUT                    / Output 
    LDA OCTAL_RESULT2      / Next digit
    ADD CONST_48           / Convert to ASCII
    OUT                    / Output
    LDA OCTAL_RESULT1      / Next digit
    ADD CONST_48           / Convert to ASCII
    OUT                    / Output
    LDA OCTAL_RESULT0      / Least significant digit
    ADD CONST_48           / Convert to ASCII
    OUT                    / Output
    HLT                    / Halt execution

/ Subroutine to divide the value in AC by 8
/ Stores Quotient in QUOTIENT_TEMP and Remainder in TEMP_REMAINDER
DIVIDE_BY_8, DEC 0     / Return address stored here
    STA TEMP_VAL       / Store the number to divide (dividend)
    CLA                / Clear AC (Quotient = 0)
    STA QUOTIENT_TEMP  / Initialize quotient to 0

DIV_LOOP, LDA TEMP_VAL     / Load current value
    ADD NEG8           / Subtract 8
    STA TEMP_VAL       / Store updated value
    SPA                / Skip if positive or zero (subtraction was successful)
    BUN DIV_DONE       / If negative, division is done

    LDA QUOTIENT_TEMP  / Load quotient
    ADD CONST_1        / Increment quotient
    STA QUOTIENT_TEMP  / Store updated quotient
    BUN DIV_LOOP       / Loop again

DIV_DONE, LDA TEMP_VAL     / Load the negative result
    ADD CONST_8        / Add 8 back to get the remainder
    STA TEMP_REMAINDER / Store the remainder
    / Quotient is already stored in QUOTIENT_TEMP
    BUN DIVIDE_BY_8 I  / Return to caller
